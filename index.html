    <script>

        // Standard-Notenschlüssel (wie ein Array von Objekten in Java)

        const defaultThresholds = [

            { name: "sehr gut+", points: 15, min: 95.0 },

            { name: "sehr gut", points: 14, min: 90.0 },

            { name: "sehr gut-", points: 13, min: 85.0 },

            { name: "gut+", points: 12, min: 80.0 },

            { name: "gut", points: 11, min: 75.0 },

            { name: "gut-", points: 10, min: 70.0 },

            { name: "befriedigend+", points: 9, min: 65.0 },

            { name: "befriedigend", points: 8, min: 60.0 },

            { name: "befriedigend-", points: 7, min: 55.0 },

            { name: "ausreichend+", points: 6, min: 50.0 },

            { name: "ausreichend", points: 5, min: 45.0 },

            { name: "ausreichend-", points: 4, min: 40.0 },

            { name: "mangelhaft+", points: 3, min: 33.0 },

            { name: "mangelhaft", points: 2, min: 27.0 },

            { name: "mangelhaft-", points: 1, min: 20.0 },

            { name: "ungenügend", points: 0, min: 0.0 }

        ];


        // Laden aus dem Speicher des Handys, falls vorhanden. Sonst Standardwerte.

        let thresholds = JSON.parse(localStorage.getItem('gradeThresholds')) || defaultThresholds;


        // Tab-Wechsel Logik

        function switchTab(tab) {

            document.getElementById('tab-calculator').classList.toggle('hidden', tab !== 'calculator');

            document.getElementById('tab-settings').classList.toggle('hidden', tab !== 'settings');

            document.querySelectorAll('.tab-btn').forEach((btn, idx) => {

                btn.classList.toggle('active', (idx === 0 && tab === 'calculator') || (idx === 1 && tab === 'settings'));

            });

            if (tab === 'settings') renderSettings();

        }


        // Tabelle in den Einstellungen zeichnen

        function renderSettings() {

            const container = document.getElementById('settings-list');

            container.innerHTML = '';

            thresholds.forEach((t, index) => {

                const row = document.createElement('div');

                row.className = 'table-row';

                row.innerHTML = `

                    <span>${t.name} (${t.points})</span>

                    <div>

                        <input type="number" step="0.1" value="${t.min}" onchange="updateThreshold(${index}, this.value)"> %

                    </div>

                `;

                container.appendChild(row);

            });

        }


        // Wert ändern und im Handy speichern

        function updateThreshold(index, value) {

            thresholds[index].min = parseFloat(value) || 0;

            localStorage.setItem('gradeThresholds', JSON.stringify(thresholds));

            calculate(); // Direkt neu berechnen im Hintergrund

        }


        // Haupt-Rechner Logik

        function calculate() {

            const max = parseFloat(document.getElementById('max-points').value);

            const achieved = parseFloat(document.getElementById('achieved-points').value);

            const resultCard = document.getElementById('result-card');


            if (max > 0 && !isNaN(achieved)) {

                resultCard.style.display = 'block';

                const percent = (achieved / max) * 100;

                document.getElementById('output-percent').innerText = `Prozent: ${percent.toFixed(2)}%`;


                // Passende Note suchen (Schleife wie in Java)

                let foundGrade = "ungenügend (0)";

                for (let t of thresholds) {

                    if (percent >= t.min) {

                        foundGrade = `${t.name} (${t.points})`;

                        break;

                    }

                }

                document.getElementById('output-grade').innerText = `Note: ${foundGrade}`;

            } else {

                resultCard.style.display = 'none';

            }

        }


        // Service Worker für Offline-Modus registrieren

        if ('serviceWorker' in navigator) {

            navigator.serviceWorker.register('sw.js');

        }

    </script>
